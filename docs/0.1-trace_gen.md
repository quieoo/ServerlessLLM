安装ServeGen
````bash
cd uv_envs/servegen
uv venv --python 3.12
source /mnt/n0/uv_envs/servegen/.venv/bin/activate

git clone https://github.com/alibaba/ServeGen.git
cd ServeGen
git checkout bdf86140c29c7fadb6317ee77b437e56d35fcf41

# replace the clientpool.py with the one in Tangram
cp Tangram/evaluation/traces/clientpool.py ServeGen/servegen/clientpool.py

uv pip install -r requirements.txt
uv pip install -e .
````
# Default
